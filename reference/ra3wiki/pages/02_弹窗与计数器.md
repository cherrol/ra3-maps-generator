首先需要在地图文件夹创建map.str文件, 在文件中保存以下内容

```lua
COUNTER:num
"选择的数字"
END
```



# 使用lua库
```lua
CounterModule.show_counter('counter', 'COUNTER:num')

local buttons = {'10', '20', '30'}
DialogModule.show_choice_dialog(1, '请选择数字', buttons, nil)

DialogModule.register_dialog_click_listener(
    function(player_name, btn_index, dialog_id)
        if dialog_id == 1 then
            CounterModule.set_counter_value('counter', tonumber(%buttons[btn_index]))           
        end
    end
)
```

# 使用原生api
```lua
ExecuteAction("DISPLAY_COUNTER", 'counter', 'COUNTER:num')

local buttons = {'10', '20', '30'}
-- 共有7个按钮, 如果不需要那么多, 需要填空字符串
exShowCustomBtnChoiceDialog(1, '请选择数字', buttons[1], buttons[2], buttons[3], '', '', '', '')

-- 注册回调函数, 函数名不可修改
function onUserBtnChoiceDialogEvent(player_name, btn_index, dialog_id)
    if dialog_id == 1 then
        exCounterSetByName('counter', tonumber(%buttons[btn_index]))           
    end
end
```
