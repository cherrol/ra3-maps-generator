# 05 顶部按钮与随机数

> 原文：https://www.yuque.com/muzeqaq/ra3mapwiki/nul3oi81aimct9qd
> 目录位置：Lua例子/代码示例

点击按钮, 随即生成一个单位;该按钮有冷却时间, 可显示冷却进度; 冷却结束时, 按钮高亮



首先在地图上新建一个路径点"Waypoint 3"



# 使用lua库
以下代码仅在第一帧运行即可

```lua
LoggerModule.setup(LoggerLevelEnum.INFO, {LoggerTargetEnum.CHAT_MSG})

Example05_meta = {
    things = {ThingEnum.AlliedCommandoTech1, ThingEnum.AlliedFighterAircraft, ThingEnum.CelestialAntiAirShip},
    button_index = 2,
    Player_1 = 1,
    Player_2 = 1
}

TopButtonModule.set_visible(1, nil)
TopButtonModule.add_or_update_button(Example05_meta.button_index, 'Button_PlayerPower_ProductionKickback', '获得随机单位', nil)


TopButtonModule.register_button_click_callback(function(player_name, button_id)
    if button_id == Example05_meta.button_index and Example05_meta[player_name] == 99 then
        ChatMsgModule.print_on_chat(player_name .. " clicked button")

        local choosed_thing = RandomHelper.choice(Example05_meta.things)
        UnitCreateHelper.create_unit_at_waypoint(
            UnitHelper.get_unit_auto_name(),
            choosed_thing,
            TeamModule.from_player_name(player_name),
            "Waypoint 3",
            nil, nil, nil
        )
        Example05_meta[player_name] = 1
    end
end)

SchedulerModule.call_every_x_frame(
    function() 
        for i = 1, 2 do
            local player_name = "Player_" .. i
            Example05_meta[player_name] = min(99, Example05_meta[player_name] + 1)

            TopButtonModule.set_button_time_overlay_value(Example05_meta.button_index, Example05_meta[player_name], player_name)
            TopButtonModule.set_button_hp_var_value(Example05_meta.button_index, Example05_meta[player_name], player_name)

            local ready = Example05_meta[player_name] == 99

            TopButtonModule.set_button_time_overlay_visible(Example05_meta.button_index, not ready, player_name)
            TopButtonModule.set_button_hp_bar_visible(Example05_meta.button_index, not ready, player_name)

            TopButtonModule.set_button_highlight(Example05_meta.button_index, ready, player_name)
            -- TopButtonModule.set_button_selected(Example05_meta.button_index, ready, player_name)
        end
    end,
 1)

```





# 使用原生api
以下代码仅在第一帧运行即可

```lua
Example05_meta = {
    things = {'AlliedCommandoTech1', 'AlliedFighterAircraft', 'CelestialAntiAirShip'},
    button_index = 2,
    Player_1 = 1,
    Player_2 = 1,
    unit_index = 1
}

exCenterTopBtnsChangeVisibility(1)
exCenterTopBtnShow(Example05_meta.button_index, 'Button_PlayerPower_ProductionKickback', '获得随机单位')

function onCenterTopBtnClickEvent(player_name, button_id)
    if button_id == Example05_meta.button_index and Example05_meta[player_name] == 99 then
        exMessageAppendToMessageArea(player_name .. " clicked button")

        local total_things_cnt = getn(Example05_meta.things)
        local random_index = 1 + floor(abs(GetRandomNumber()) * total_things_cnt)
        exMessageAppendToMessageArea('random_index: ' .. tostring(random_index))
        local choosed_thing = Example05_meta.things[random_index]
        exMessageAppendToMessageArea('choosed_thing: ' .. tostring(choosed_thing))
        ExecuteAction("CREATE_NAMED_ON_TEAM_AT_WAYPOINT", 
        "created_unit" .. tostring(Example05_meta.unit_index), 
        choosed_thing, 
        player_name .. "/team" .. player_name,
         "Waypoint 3")

        Example05_meta[player_name] = 1

        Example05_meta.unit_index = Example05_meta.unit_index + 1
    end
end
```



以下代码需要每帧运行

```lua

for i = 1, 2 do
    local player_name = "Player_" .. i
    Example05_meta[player_name] = min(99, Example05_meta[player_name] + 1)
    exCenterTopBtnSetTimerOverlayValueForPlayer(player_name, Example05_meta.button_index, Example05_meta[player_name])
    exCenterTopBtnSetHealthBarValueForPlayer(player_name, Example05_meta.button_index, Example05_meta[player_name])

    if Example05_meta[player_name] == 99 then
        exCenterTopBtnSetTimerOverlayVisibilityForPlayer(player_name, Example05_meta.button_index, 0)
        exCenterTopBtnSetHealthBarVisibilityForPlayer(player_name, Example05_meta.button_index, 0)
        exCenterTopBtnToggleHighLightForPlayer(player_name, Example05_meta.button_index, 1)
    else
        exCenterTopBtnSetTimerOverlayVisibilityForPlayer(player_name, Example05_meta.button_index, 1)
        exCenterTopBtnSetHealthBarVisibilityForPlayer(player_name, Example05_meta.button_index, 1)
        exCenterTopBtnToggleHighLightForPlayer(player_name, Example05_meta.button_index, 0)
    end
end
```
