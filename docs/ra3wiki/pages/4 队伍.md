# 队伍
```python
from ra3map import Ra3Map

m_2 = Ra3Map.load_map('m_2')

# 获取所有队伍信息
# 这三个数据分别为 所属玩家名, 队伍名, 队伍全名
# 如: 
# Player_1, teamPlayer_1, Player_1/teamPlayer_1
# Player_2, t1, Player_2/t1
for t in m.get_teams():
    print(t.belong_to_player_name, t.team_name,t.team_full_name)

# 新建一个队伍
t = m_2.add_team('t1', 'Player_2')
# 可以修改队伍信息
t.belong_to_player_name = 'Player_3'
t.team_name = 't3'

# 删除队伍
m_2.remove_team('t3', 'Player_3')

# 别忘了保存
m_2.save()
```
